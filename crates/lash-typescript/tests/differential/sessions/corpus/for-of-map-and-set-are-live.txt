const counts = new Map([['a', 1], ['b', 2], ['c', 3]]);
for (const [key, value] of counts) {
  if (key === 'a') {
    counts.delete('b');
    counts.set('d', 4);
    counts.set('c', 30);
  }
  console.log(key, value);
}
const tags = new Set(['x']);
for (const tag of tags) {
  if (tags.size < 3) {
    tags.add(tag + 'y');
  }
  console.log(tag);
}
