const out = [];
{
  const map = new Map([['k', 1], ['j', 2]]);
  const alias = map;
  alias.forEach((value, key) => { out.push(key + value); });
  const box = { set: new Set([4, 5]), params: new URLSearchParams('a=1') };
  box.set.forEach((value) => { out.push(value * 2); });
  box.params.forEach((value, key) => { out.push(key); });
  console.log([...alias.keys()], [...alias.values()], [...box.params.entries()]);
}
console.log(out);
